## 三、详细任务清单

### 阶段一：基础框架搭建

#### 任务 1.1：后端项目初始化

**目标：** 搭建 Node.js + Express + Sequelize 后端框架

**步骤：**
1. 创建项目目录 `backend/`
2. 初始化 package.json
3. 安装依赖：
   - express
   - sequelize
   - mysql2
   - dotenv
   - cors
   - bcryptjs
   - jsonwebtoken
   - multer
   - archiver（打包）
   - exceljs（Excel导出）
4. 创建目录结构：
   ```
   backend/
   ├── src/
   │   ├── app.js
   │   ├── config/
   │   ├── controllers/
   │   ├── middleware/
   │   ├── models/
   │   ├── routes/
   │   ├── services/
   │   └── utils/
   ├── uploads/
   ├── tests/
   ├── package.json
   └── .env
   ```
5. 配置数据库连接
6. 创建基础 Express 应用

**验收标准：**
- [ ] 项目可正常启动
- [ ] 数据库连接成功
- [ ] 基础路由可访问

---

#### 任务 1.2：前端项目初始化

**目标：** 搭建 Vue3 + Vite + Element Plus 前端框架

### 4.1 订单编号生成

```javascript
// 示例代码
async function generateOrderNo(groupId) {
  const group = await Group.findByPk(groupId, {
    include: [{ model: District, include: [Province] }]
  });
  
  const now = new Date();
  const yearMonth = `${now.getFullYear().toString().slice(-2)}${(now.getMonth() + 1).toString().padStart(2, '0')}`;
  
  // 查询当月该小组最大流水号
  const lastOrder = await Order.findOne({
    where: {
      group_id: groupId,
      created_at: { [Op.gte]: new Date(now.getFullYear(), now.getMonth(), 1) }
    },
    order: [['order_no', 'DESC']]
  });
  
  let serial = 1;
  if (lastOrder) {
    const lastSerial = parseInt(lastOrder.order_no.split('-')[4]);
    serial = lastSerial + 1;
  }
  
  return `${group.district.province.code}-${group.district.code}-${group.code}-${yearMonth}-${serial.toString().padStart(4, '0')}`;
}
```

### 4.3 批量导出CDR

```javascript
const archiver = require('archiver');
const exceljs = require('exceljs');

async function exportCDR(faceIds) {
  const faces = await MeasureFace.findAll({
    where: { id: faceIds },
    include: [/* 关联查询 */]
  });
  
  // 创建ZIP
  const archive = archiver('zip');
  
  // 添加CDR文件
  for (const face of faces) {
    const cdrFile = face.cdrFile;
    if (cdrFile) {
      archive.file(cdrFile.file_path, { name: `${face.order.order_no}_${face.face_name}.cdr` });
    }
  }
  
  // 创建Excel清单
  const workbook = new exceljs.Workbook();
  const worksheet = workbook.addWorksheet('生产清单');
  // 添加数据...
  
  const excelBuffer = await workbook.xlsx.writeBuffer();
  archive.append(excelBuffer, { name: '生产清单.xlsx' });
  
  archive.finalize();
  return archive;
}
```
